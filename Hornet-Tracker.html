    <script>

        const SERVICE_UUID = "4fafc201-1fb5-459e-8fcc-c5c9c331914b";

        const TX_CHARACTERISTIC_UUID = "1b4e28bf-2baf-4253-93ec-d368e19a6d01";

        const RX_CHARACTERISTIC_UUID = "beb5483e-36e1-4688-b7f5-ea07361b26a8";


        let gattServer = null;

        let rxCharacteristic = null;

        let txCharacteristic = null;

        let csvAccumulateur = "";


        function log(msg) {

            const cons = document.getElementById("console");

            cons.innerHTML += msg + "<br>";

            cons.scrollTop = cons.scrollHeight;

        }


        async function connectBLE() {

            try {

                log("Recherche du tracker...");

                const device = await navigator.bluetooth.requestDevice({

                    filters: [{ name: 'Heltec_Frelon_Tracker' }],

                    optionalServices: [SERVICE_UUID]

                });


                log("Connexion à l'appareil...");

                gattServer = await device.gatt.connect();

                

                const service = await gattServer.getPrimaryService(SERVICE_UUID);

                rxCharacteristic = await service.getCharacteristic(RX_CHARACTERISTIC_UUID);

                txCharacteristic = await service.getCharacteristic(TX_CHARACTERISTIC_UUID);


                // Activer l'écoute (Notifications)

                await txCharacteristic.startNotifications();

                txCharacteristic.addEventListener('characteristicvaluechanged', gererDonneesRecues);


                document.getElementById("bleStatus").innerText = "Connecté !";

                document.getElementById("bleStatus").className = "status connected";

                

                // Activer les boutons

                document.getElementById("btnSync").disabled = false;

                document.getElementById("btnRead").disabled = false;

                document.getElementById("btnClear").disabled = false;

                

                log("Système Prêt !");

            } catch (error) {

                log("Erreur : " + error);

            }

        }


        function gererDonneesRecues(event) {

            let enc = new TextDecoder("utf-8");

            let texte = enc.decode(event.target.value);

            

            if (texte.includes("--- DEBUT EXPORT CSV ---")) {

                csvAccumulateur = "";

                log("Téléchargement commencé...");

                return;

            }

            

            if (texte.includes("--- FIN EXPORT CSV ---")) {

                log("Téléchargement Terminé ! Création du fichier...");

                telechargerFichierCSV(csvAccumulateur);

                return;

            }


            if(!texte.startsWith("INFO:")) {

                csvAccumulateur += texte + "\n";

            }

            log(texte);

        }


        async function synchroniserHeure() {

            const maintenant = new Date();

            const pad = (n) => String(n).padStart(2, '0');

            const dateStr = `${maintenant.getFullYear()}-${pad(maintenant.getMonth()+1)}-${pad(maintenant.getDate())} ${pad(maintenant.getHours())}:${pad(maintenant.getMinutes())}:${pad(maintenant.getSeconds())}`;

            

            let enc = new TextEncoder();

            await rxCharacteristic.writeValue(enc.encode(dateStr));

            log("Heure envoyée : " + dateStr);

        }


        async function demanderDonnees() {

            let enc = new TextEncoder();

            await rxCharacteristic.writeValue(enc.encode("LIRE"));

            log("Demande d'exportation envoyée...");

        }


        async function viderMemoire() {

            if (confirm("Voulez-vous vraiment effacer définitivement les logs stockés sur la carte Heltec ?")) {

                let enc = new TextEncoder();

                await rxCharacteristic.writeValue(enc.encode("VIDER"));

                log("Ordre d'effacement envoyé.");

            }

        }


        function telechargerFichierCSV(contenu) {

            const blob = new Blob([contenu], { type: 'text/csv;charset=utf-8;' });

            const url = URL.createObjectURL(blob);

            const a = document.createElement("a");

            a.href = url;

            a.download = `logs_frelons_${new Date().toISOString().substring(0,10)}.csv`;

            document.body.appendChild(a);

            a.click();

            document.body.removeChild(a);

        }
